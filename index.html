7. Working test rig (PowerShell, no install needed)

function Try-Pass {
    param([string]$Pass, [string]$File = "cosmic.txt")
    $b64 = (Get-Content $File -Raw) -replace '\s',''
    $raw = [Convert]::FromBase64String($b64)
    $salt = $raw[8..15]; $ct = $raw[16..($raw.Length-1)]
    $pb = [Text.Encoding]::UTF8.GetBytes($Pass)
    $sha = [Security.Cryptography.SHA256]::Create()
    $data=@(); $prev=@()
    while ($data.Length -lt 48){ $prev=$sha.ComputeHash($prev+$pb+$salt); $data+=$prev }
    $aes=[Security.Cryptography.Aes]::Create(); $aes.Mode='CBC'; $aes.Padding='None'
    $aes.Key=$data[0..31]; $aes.IV=$data[32..47]
    $plain=$aes.CreateDecryptor().TransformFinalBlock($ct,0,$ct.Length)
    $printable=($plain | Where-Object {$_ -ge 32 -and $_ -lt 127}).Count / $plain.Length
    if ($printable -gt 0.85){ Write-Host "*** HIT pass='$Pass'" -ForegroundColor Green; Write-Host ([Text.Encoding]::UTF8.GetString($plain)) }
    else { Write-Host "no ($([math]::Round($printable*100))% printable)" }
}
function SHA256Hex($s){ -join ([Security.Cryptography.SHA256]::Create().ComputeHash([Text.Encoding]::UTF8.GetBytes($s)) | %{ $_.ToString('x2') }) }
